"""The two tool schemas this fixture's model call binds. Invented; nothing here is read from a checkout.

Each class holds its own JSON schema as a literal in its body, which is how a great many
applications write a tool: the class is what the application refers to, the literal is what goes on
the wire, and both are text a resolver can find.
"""


class ProposeRoute:
    """The submission tool: the route the model proposes."""

    NAME = "propose_route"
    SCHEMA = {
        "name": "propose_route",
        "description": "Propose one delivery route.",
        "input_schema": {
            "type": "object",
            "required": ["depot", "stops", "grade"],
            "properties": {
                "depot": {"type": "string", "minLength": 3},
                "stops": {"type": "array", "maxItems": 5, "minItems": 1},
                "grade": {"type": "string", "enum": ["standard", "express", "overnight"]},
                "distance_km": {"type": "number", "maximum": 400},
            },
        },
    }


class CheckCapacity:
    """The lookup tool: how much the depot can still take today."""

    NAME = "check_capacity"
    SCHEMA = {
        "name": "check_capacity",
        "description": "Ask the depot how much capacity is left.",
        "input_schema": {
            "type": "object",
            "required": ["depot"],
            "properties": {
                "depot": {"type": "string", "minLength": 3},
                "window_hours": {"type": "integer", "maximum": 24, "minimum": 1},
            },
        },
    }
